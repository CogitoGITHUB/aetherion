# cd + custom logic based on the value of _ZO_ECHO.
fn __aiu_shell_jump_cd {|path|
    builtin:cd $path
{%- if echo %}
    builtin:echo $pwd
{%- endif %}
}

# When using aiu-shell-jump with --no-cmd, alias these internal functions as desired.
#

# Jump to a directory using only keywords.
fn __aiu_shell_jump_z {|@rest|
    if (builtin:eq [] $rest) {
        __aiu_shell_jump_cd ~
    } elif (builtin:eq [-] $rest) {
        __aiu_shell_jump_cd $oldpwd
    } elif (and ('builtin:==' (builtin:count $rest) 1) (path:is-dir &follow-symlink=$true $rest[0])) {
        __aiu_shell_jump_cd $rest[0]
    } else {
        var path
        try {
            set path = (aiu-shell-jump query --exclude $pwd -- $@rest)
        } catch {
        } else {
            __aiu_shell_jump_cd $path
        }
    }
}
edit:add-var __aiu_shell_jump_z~ $__aiu_shell_jump_z~

# Jump to a directory using interactive search.
fn __aiu_shell_jump_zi {|@rest|
    var path
    try {
        set path = (aiu-shell-jump query --interactive -- $@rest)
    } catch {
    } else {
        __aiu_shell_jump_cd $path
    }
}
edit:add-var __aiu_shell_jump_zi~ $__aiu_shell_jump_zi~

# Load completions.
{#-
  aiu-shell-jump-based completions are currently not possible, because Elvish only prints
  a completion if the current token is a prefix of it.
#}
fn __aiu_shell_jump_z_complete {|@rest|
    if (!= (builtin:count $rest) 2) {
        builtin:return
    }
    edit:complete-filename $rest[1] |
        builtin:each {|completion|
            var dir = $completion[stem]
            if (path:is-dir $dir) {
                builtin:put $dir
            }
        }
}
set edit:completion:arg-completer[{{cmd}}] = $__aiu_shell_jump_z_complete~
