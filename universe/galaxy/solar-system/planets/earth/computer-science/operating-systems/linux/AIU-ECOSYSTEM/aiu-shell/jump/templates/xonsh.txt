# Utility functions for aiu-shell-jump.
#


def __aiu_shell_jump_bin() -> str:
    """Finds and returns the location of the aiu-shell-jump binary."""
    aiu-shell-jump = typing.cast(str, xonsh.environ.locate_binary("aiu-shell-jump"))
    if aiu-shell-jump is None:
        aiu-shell-jump = "aiu-shell-jump"
    return aiu-shell-jump


def __aiu_shell_jump_env() -> dict[str, str]:
    """Returns the current environment."""
    return builtins.__xonsh__.env.detype()  # type: ignore  # pylint:disable=no-member


def __aiu_shell_jump_pwd() -> str:
    """pwd based on the value of _ZO_RESOLVE_SYMLINKS."""
{%- if resolve_symlinks %}
    pwd = os.getcwd()
{%- else %}
    pwd = __aiu_shell_jump_env().get("PWD")
    if pwd is None:
        raise RuntimeError("$PWD not found")
{%- endif %}
    return pwd


def __aiu_shell_jump_cd(path: str | bytes | None = None) -> None:
    """cd + custom logic based on the value of _ZO_ECHO."""
    if path is None:
        args = []
    elif isinstance(path, bytes):
        args = [path.decode("utf-8")]
    else:
        args = [path]
    _, exc, _ = xonsh.dirstack.cd(args)
    if exc is not None:
        raise RuntimeError(exc)
{%- if echo %}
    print(__aiu_shell_jump_pwd())
{%- endif %}


class AiuShellJumpSilentException(Exception):
    """Exit without complaining."""


def __aiu_shell_jump_errhandler(
    func: typing.Callable[[list[str]], None],
) -> typing.Callable[[list[str]], int]:
    """Print exception and exit with error code 1."""

    def wrapper(args: list[str]) -> int:
        try:
            func(args)
            return 0
        except AiuShellJumpSilentException:
            return 1
        except Exception as exc:  # pylint: disable=broad-except
            print(f"aiu-shell-jump: {exc}", file=sys.stderr)
            return 1

    return wrapper

@__aiu_shell_jump_errhandler
def __aiu_shell_jump_z(args: list[str]) -> None:
    """Jump to a directory using only keywords."""
    if args == []:
        __aiu_shell_jump_cd()
    elif args == ["-"]:
        __aiu_shell_jump_cd("-")
    elif len(args) == 1 and os.path.isdir(args[0]):
        __aiu_shell_jump_cd(args[0])
    else:
        try:
            aiu-shell-jump = __aiu_shell_jump_bin()
            cmd = subprocess.run(
                [aiu-shell-jump, "query", "--exclude", __aiu_shell_jump_pwd(), "--"] + args,
                check=True,
                env=__aiu_shell_jump_env(),
                stdout=subprocess.PIPE,
            )
        except subprocess.CalledProcessError as exc:
            raise AiuShellJumpSilentException() from exc

        result = cmd.stdout[:-1]
        __aiu_shell_jump_cd(result)


@__aiu_shell_jump_errhandler
def __aiu_shell_jump_zi(args: list[str]) -> None:
    """Jump to a directory using interactive search."""
    try:
        aiu-shell-jump = __aiu_shell_jump_bin()
        cmd = subprocess.run(
            [aiu-shell-jump, "query", "-i", "--"] + args,
            check=True,
            env=__aiu_shell_jump_env(),
            stdout=subprocess.PIPE,
        )
    except subprocess.CalledProcessError as exc:
        raise AiuShellJumpSilentException() from exc

    result = cmd.stdout[:-1]
    __aiu_shell_jump_cd(result)
