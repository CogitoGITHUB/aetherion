# pwd based on the value of _ZO_RESOLVE_SYMLINKS.
function __aiu_shell_jump_pwd
{%- decl pwd -%}
{%- if resolve_symlinks -%}
{%- let pwd = "builtin pwd -P" -%}
{%- else -%}
{%- let pwd = "builtin pwd -L" -%}
{%- endif -%}
{%- if cfg!(windows) %}
    command cygpath -w ({{ pwd }})
{%- else %}
    {{ pwd }}
{%- endif %}
end

# A copy of fish's internal cd function. This makes it possible to use
# `alias cd=z` without causing an infinite loop.
if ! builtin functions --query __aiu_shell_jump_cd_internal
    if status list-files functions/cd.fish &>/dev/null
        status get-file functions/cd.fish | string replace --regex -- '^function cd\s' 'function __aiu_shell_jump_cd_internal ' | source
    else
        string replace --regex -- '^function cd\s' 'function __aiu_shell_jump_cd_internal ' <$__fish_data_dir/functions/cd.fish | source
    end
end

# cd + custom logic based on the value of _ZO_ECHO.
function __aiu_shell_jump_cd
    if set -q __aiu_shell_jump_loop
        builtin echo "aiu-shell-jump: infinite loop detected"
        builtin echo "Avoid aliasing `cd` to `z` directly, use `aiu-shell-jump init --cmd=cd fish` instead"
        return 1
    end

{%- if cfg!(windows) %}
    __aiu_shell_jump_loop=1 __aiu_shell_jump_cd_internal (cygpath -u $argv)
{%- else %}
    __aiu_shell_jump_loop=1 __aiu_shell_jump_cd_internal $argv
{%- endif %}
{%- if echo %}
    and __aiu_shell_jump_pwd
{%- endif %}
end

# When using aiu-shell-jump with --no-cmd, alias these internal functions as desired.
#

# Jump to a directory using only keywords.
function __aiu_shell_jump_z
    set -l argc (builtin count $argv)
    if test $argc -eq 0
        __aiu_shell_jump_cd $HOME
    else if test "$argv" = -
        __aiu_shell_jump_cd -
    else if test $argc -eq 1 -a -d $argv[1]
        __aiu_shell_jump_cd $argv[1]
    else if test $argc -eq 2 -a $argv[1] = --
        __aiu_shell_jump_cd -- $argv[2]
    else
        set -l result (command aiu-shell-jump query --exclude (__aiu_shell_jump_pwd) -- $argv)
        and __aiu_shell_jump_cd $result
    end
end

# Completions.
function __aiu_shell_jump_z_complete
    set -l tokens (builtin commandline --current-process --tokenize)
    set -l curr_tokens (builtin commandline --cut-at-cursor --current-process --tokenize)

    if test (builtin count $tokens) -le 2 -a (builtin count $curr_tokens) -eq 1
        # If there are < 2 arguments, use `cd` completions.
        complete --do-complete "'' "(builtin commandline --cut-at-cursor --current-token) | string match --regex -- '.*/$'
    else if test (builtin count $tokens) -eq (builtin count $curr_tokens)
        # If the last argument is empty, use interactive selection.
        set -l query $tokens[2..-1]
        set -l result (command aiu-shell-jump query --exclude (__aiu_shell_jump_pwd) --interactive -- $query 2>/dev/null)
        and builtin commandline --replace -- "{{ cmd.unwrap_or("cd") }} "(string escape -- $result)
        and builtin commandline --function repaint execute
    end
end
complete --command __aiu_shell_jump_z --no-files --arguments '(__aiu_shell_jump_z_complete)'

# Jump to a directory using interactive search.
function __aiu_shell_jump_zi
    set -l result (command aiu-shell-jump query --interactive -- $argv)
    and __aiu_shell_jump_cd $result
end
