# Utility functions for aiu-shell-jump.
#

# Call aiu-shell-jump binary, returning the output as UTF-8.
function global:__aiu_shell_jump_bin {
    $encoding = [Console]::OutputEncoding
    try {
        [Console]::OutputEncoding = [System.Text.Utf8Encoding]::new()
        $result = aiu-shell-jump @args
        return $result
    } finally {
        [Console]::OutputEncoding = $encoding
    }
}

# pwd based on aiu-shell-jump's format.
function global:__aiu_shell_jump_pwd {
    $cwd = Microsoft.PowerShell.Management\Get-Location
    if ($cwd.Provider.Name -eq "FileSystem") {
        $cwd.ProviderPath
    }
}

# cd + custom logic based on the value of _ZO_ECHO.
function global:__aiu_shell_jump_cd($dir, $literal) {
    $dir = if ($literal) {
        if ($null -eq $dir) {
            Microsoft.PowerShell.Management\Set-Location
        } else {
            Microsoft.PowerShell.Management\Set-Location -LiteralPath $dir -Passthru -ErrorAction Stop
        }
    } else {
        if ($dir -eq '-' -and ($PSVersionTable.PSVersion -lt 6.1)) {
            Microsoft.PowerShell.Utility\Write-Error "cd - is not supported below PowerShell 6.1. Please upgrade your version of PowerShell."
        }
        elseif ($dir -eq '+' -and ($PSVersionTable.PSVersion -lt 6.2)) {
            Microsoft.PowerShell.Utility\Write-Error "cd + is not supported below PowerShell 6.2. Please upgrade your version of PowerShell."
        }
        else {
            Microsoft.PowerShell.Management\Set-Location -Path $dir -Passthru -ErrorAction Stop
        }
    }
{%- if echo %}
    Microsoft.PowerShell.Utility\Write-Output $dir.Path
{%- endif %}
}

        aiu-shell-jump add "--" $result

    $result = __aiu_shell_jump_pwd

# Jump to a directory using only keywords.
function global:__aiu_shell_jump_z {
    if ($args.Length -eq 0) {
        __aiu_shell_jump_cd $null $true
    }
    elseif ($args.Length -eq 1 -and ($args[0] -eq '-' -or $args[0] -eq '+')) {
        __aiu_shell_jump_cd $args[0] $false
    }
    elseif ($args.Length -eq 1 -and (Microsoft.PowerShell.Management\Test-Path -PathType Container -LiteralPath $args[0])) {
        __aiu_shell_jump_cd $args[0] $true
    }
    elseif ($args.Length -eq 1 -and (Microsoft.PowerShell.Management\Test-Path -PathType Container -Path $args[0] )) {
        __aiu_shell_jump_cd $args[0] $false
    }
    else {
        $result = __aiu_shell_jump_pwd
        if ($null -ne $result) {
            $result = __aiu_shell_jump_bin query --exclude $result "--" @args
        }
        else {
            $result = __aiu_shell_jump_bin query "--" @args
        }
        if ($LASTEXITCODE -eq 0) {
            __aiu_shell_jump_cd $result $true
        }
    }
}

# Jump to a directory using interactive search.
function global:__aiu_shell_jump_zi {
    $result = __aiu_shell_jump_bin query -i "--" @args
    if ($LASTEXITCODE -eq 0) {
        __aiu_shell_jump_cd $result $true
    }
}
