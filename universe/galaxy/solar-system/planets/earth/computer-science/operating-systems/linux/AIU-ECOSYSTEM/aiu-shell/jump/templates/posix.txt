# pwd based on the value of _ZO_RESOLVE_SYMLINKS.
{%- decl pwd -%}
{%- if resolve_symlinks -%}
{%- let pwd = "\\command pwd -P" -%}
{%- else -%}
{%- let pwd = "\\command pwd -L" -%}
{%- endif -%}
{%- if cfg!(windows) %}
if \command -v cygpath >/dev/null
then
  __aiu_shell_jump_pwd() {
    \command cygpath -w "$({{ pwd }})"
  }
else
  __aiu_shell_jump_pwd() {
    {{ pwd }}
  }
fi
{%- else %}
__aiu_shell_jump_pwd() {
    {{ pwd }}
}
{%- endif %}

# cd + custom logic based on the value of _ZO_ECHO.
__aiu_shell_jump_cd() {
    # shellcheck disable=SC2164
    \command cd "$@" {%- if echo %} && __aiu_shell_jump_pwd {%- endif %}
}

{% match hook %}

{%- when InitHook::Prompt -%}
# Hook to add new entries to the database.
__aiu_shell_jump_hook() {
    \command aiu-shell-jump add -- "$(__aiu_shell_jump_pwd || \command true)"
}

# Initialize hook.
if [ "${PS1:=}" = "${PS1#*\$(__aiu_shell_jump_hook)}" ]; then
    PS1="${PS1}\$(__aiu_shell_jump_hook)"
fi

# Report common issues.
__aiu_shell_jump_doctor() {
{%- if hook != InitHook::Prompt %}
    return 0
{%- else %}
    [ "${_ZO_DOCTOR:-1}" -eq 0 ] && return 0
    case "${PS1:-}" in
    *__aiu_shell_jump_hook*) return 0 ;;
    *) ;;
    esac

    _ZO_DOCTOR=0
    \command printf '%s\n' \
        'aiu-shell-jump: detected a possible configuration issue.' \
        'Please ensure that aiu-shell-jump is initialized right at the end of your shell configuration file.' \
        '' \
        'If the issue persists, consider filing an issue at:' \
        'https://github.com/ajeetdsouza/zoxide/issues' \
        '' \
        'Disable this message by setting _ZO_DOCTOR=0.' \
        '' >&2
{%- endif %}
}

# When using aiu-shell-jump with --no-cmd, alias these internal functions as desired.
#

# Jump to a directory using only keywords.
__aiu_shell_jump_z() {
    __aiu_shell_jump_doctor

    if [ "$#" -eq 0 ]; then
        __aiu_shell_jump_cd ~
    elif [ "$#" -eq 1 ] && [ "$1" = '-' ]; then
        if [ -n "${OLDPWD}" ]; then
            __aiu_shell_jump_cd "${OLDPWD}"
        else
            # shellcheck disable=SC2016
            \command printf 'aiu-shell-jump: $OLDPWD is not set'
            return 1
        fi
    elif [ "$#" -eq 1 ] && (\command cd -- "$1") >/dev/null 2>&1; then
        __aiu_shell_jump_cd "$1"
    else
        __aiu_shell_jump_result="$(\command aiu-shell-jump query --exclude "$(__aiu_shell_jump_pwd || \command true)" -- "$@")" &&
            __aiu_shell_jump_cd "${__aiu_shell_jump_result}"
    fi
}

# Jump to a directory using interactive search.
__aiu_shell_jump_zi() {
    __aiu_shell_jump_doctor
    __aiu_shell_jump_result="$(\command aiu-shell-jump query --interactive -- "$@")" && __aiu_shell_jump_cd "${__aiu_shell_jump_result}"
}
